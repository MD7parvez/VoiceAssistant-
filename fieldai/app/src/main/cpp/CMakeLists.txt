cmake_minimum_required(VERSION 3.22.1)
project(fieldai_jni LANGUAGES C CXX)
if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/llama.cpp/CMakeLists.txt")
    message(FATAL_ERROR "llama.cpp source missing. GitHub Actions workflow must fetch the source first.")
endif()
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_VULKAN OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
add_subdirectory(llama.cpp)
add_library(fieldai_llama SHARED fieldai_llama.cpp)
target_compile_features(fieldai_llama PRIVATE cxx_std_17)
target_include_directories(fieldai_llama PRIVATE llama.cpp/include llama.cpp/ggml/include)
target_link_libraries(fieldai_llama PRIVATE llama android log)
